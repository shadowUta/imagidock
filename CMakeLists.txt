cmake_minimum_required(VERSION 3.24)
project(ImagiDock VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(MSVC)
  add_compile_options(/bigobj /MP /Zc:__cplusplus)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
set(IMAGIDOCK_QT "${CMAKE_SOURCE_DIR}/.tools/qt/6.8.3/msvc2022_64" CACHE PATH "Qt installation")
list(PREPEND CMAKE_PREFIX_PATH "${IMAGIDOCK_QT}")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent Multimedia MultimediaWidgets Test)
add_library(sqlite3 STATIC .tools/sqlite/sqlite-amalgamation-3500400/sqlite3.c)
target_include_directories(sqlite3 PUBLIC .tools/sqlite/sqlite-amalgamation-3500400)
target_compile_definitions(sqlite3 PRIVATE SQLITE_ENABLE_FTS5 SQLITE_THREADSAFE=1 SQLITE_ENABLE_COLUMN_METADATA)
set_target_properties(sqlite3 PROPERTIES AUTOMOC OFF)
file(GLOB_RECURSE ZSTD_SOURCES CONFIGURE_DEPENDS .tools/zstd/zstd-1.5.7/lib/common/*.c .tools/zstd/zstd-1.5.7/lib/compress/*.c .tools/zstd/zstd-1.5.7/lib/decompress/*.c .tools/zstd/zstd-1.5.7/lib/dictBuilder/*.c)
add_library(zstd STATIC ${ZSTD_SOURCES})
target_include_directories(zstd PUBLIC .tools/zstd/zstd-1.5.7/lib)
set_target_properties(zstd PROPERTIES AUTOMOC OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL_PDB OFF CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_EXPORTER ON CACHE BOOL "" FORCE)
add_subdirectory(.tools/assimp/assimp-5.4.3 EXCLUDE_FROM_ALL)
set(FILAMENT_ROOT "${CMAKE_SOURCE_DIR}/.tools/filament" CACHE PATH "Filament SDK")
set(FILAMENT_LIBS gltfio_core gltfio filament backend filabridge filaflat utils geometry shaders uberarchive uberzlib stb ktxreader basis_transcoder dracodec meshoptimizer mikktspace bluegl bluevk smol-v image filamat matdbg abseil zstd)
add_library(filament_sdk INTERFACE)
target_include_directories(filament_sdk INTERFACE "${FILAMENT_ROOT}/include")
target_link_directories(filament_sdk INTERFACE "${FILAMENT_ROOT}/lib/x86_64/md")
foreach(lib IN LISTS FILAMENT_LIBS)
  target_link_libraries(filament_sdk INTERFACE "${lib}.lib")
endforeach()
target_link_libraries(filament_sdk INTERFACE opengl32 gdi32 user32 shlwapi ws2_32)
if(EXISTS "${CMAKE_SOURCE_DIR}/src/render/ModelView.cpp")
  add_library(imagidock_render STATIC src/render/ModelView.cpp src/render/ModelView.h src/render/RenderScene.cpp src/render/RenderScene.h src/render/ModelPrepare.cpp src/render/ModelPrepare.h)
  target_link_libraries(imagidock_render PUBLIC Qt6::Widgets Qt6::Concurrent filament_sdk assimp)
  target_include_directories(imagidock_render PUBLIC src)
  add_executable(imagidock-render src/render/render_main.cpp)
  target_link_libraries(imagidock-render PRIVATE imagidock_render)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/src/core/Library.cpp")
  file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/core/*.h)
  list(REMOVE_ITEM CORE_SOURCES "${CMAKE_SOURCE_DIR}/src/core/service_main.cpp")
  add_library(imagidock_core STATIC ${CORE_SOURCES})
  target_include_directories(imagidock_core PUBLIC src)
  target_link_libraries(imagidock_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Concurrent sqlite3 zstd)
  add_executable(imagidock-service src/core/service_main.cpp)
  target_link_libraries(imagidock-service PRIVATE imagidock_core)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/src/ui/main.cpp")
  file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.h)
  add_executable(ImagiDock WIN32 ${UI_SOURCES})
  target_include_directories(ImagiDock PRIVATE src)
  target_link_libraries(ImagiDock PRIVATE Qt6::Widgets Qt6::Network Qt6::Concurrent Qt6::Multimedia Qt6::MultimediaWidgets imagidock_render)
  if(WIN32)
    target_link_libraries(ImagiDock PRIVATE d3d11 dxgi)
  endif()
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/helpers/sai1_thumbnail.cpp")
  add_subdirectory(.tools/sources/libsai EXCLUDE_FROM_ALL)
  add_executable(imagidock-sai1 helpers/sai1_thumbnail.cpp)
  target_link_libraries(imagidock-sai1 PRIVATE Qt6::Gui sai)
endif()
enable_testing()
if(EXISTS "${CMAKE_SOURCE_DIR}/tests/inspector_tests.cpp")
  add_executable(imagidock-inspector-tests tests/inspector_tests.cpp src/ui/InspectorSummary.cpp src/ui/InspectorSummary.h src/ui/Theme.cpp src/ui/UiIcons.cpp src/ui/UiStyle.cpp src/ui/UiStyle.h)
  target_include_directories(imagidock-inspector-tests PRIVATE src)
  target_link_libraries(imagidock-inspector-tests PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME inspector COMMAND imagidock-inspector-tests)
  set_tests_properties(inspector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/tests/grid_tests.cpp")
  add_executable(imagidock-grid-tests tests/grid_tests.cpp src/ui/AssetGrid.cpp src/ui/AssetGrid.h src/ui/RpcClient.cpp src/ui/RpcClient.h src/ui/Theme.cpp src/ui/UiStyle.cpp src/ui/UiStyle.h)
  target_include_directories(imagidock-grid-tests PRIVATE src)
  target_link_libraries(imagidock-grid-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Concurrent Qt6::Test)
  add_test(NAME grid COMMAND imagidock-grid-tests)
  set_tests_properties(grid PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
add_executable(imagidock-media helpers/media_preview.cpp)
target_link_libraries(imagidock-media PRIVATE Qt6::Gui Qt6::Multimedia)
if(EXISTS "${CMAKE_SOURCE_DIR}/tests/core_tests.cpp")
  add_executable(imagidock-core-tests tests/core_tests.cpp)
  target_link_libraries(imagidock-core-tests PRIVATE imagidock_core Qt6::Test)
  add_test(NAME core COMMAND imagidock-core-tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_LOGGING_RULES=qt.multimedia.*=false")
endif()
